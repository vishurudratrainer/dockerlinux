Use code with caution.4. The Dockerfile (Dockerfile)Create a file named Dockerfile (no file extension). This uses the industry-standard multi-stage build process.dockerfile# STAGE 1: Compile the application using the SDK
FROM ://microsoft.com AS build-env
WORKDIR /app

# Copy the project file and restore dependencies
COPY *.csproj ./
RUN dotnet restore

# Copy the source code and compile the release binaries
COPY . ./
RUN dotnet publish -c Release -o out

# STAGE 2: Run the application using the lightweight Runtime
FROM ://microsoft.com
WORKDIR /app

# Copy compiled files from Stage 1
COPY --from=build-env /app/out .

# Expose the default .NET container port
EXPOSE 8080

# Start the application
ENTRYPOINT ["dotnet", "StudentApi.dll"]
